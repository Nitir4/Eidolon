add_library(or_equations STATIC equations.cpp)
add_library(tora::equations ALIAS or_equations)
tora_target_defaults(or_equations)
target_compile_features(or_equations PUBLIC cxx_std_17)
target_include_directories(or_equations PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(or_equations PUBLIC tora::core)
if(MSVC)
    target_compile_options(or_equations PRIVATE /fp:strict)
else()
    target_compile_options(or_equations PRIVATE -ffp-contract=off -fno-fast-math)
endif()
