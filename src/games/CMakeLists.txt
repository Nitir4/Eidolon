add_library(or_games STATIC games.cpp graphical.cpp)
add_library(tora::games ALIAS or_games)
tora_target_defaults(or_games)
target_compile_features(or_games PUBLIC cxx_std_17)
target_include_directories(or_games PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(or_games PUBLIC tora::lp)
if(MSVC)
    target_compile_options(or_games PRIVATE /fp:strict)
else()
    target_compile_options(or_games PRIVATE -ffp-contract=off -fno-fast-math)
endif()
